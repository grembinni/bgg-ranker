```
MAX_GAMES: 990  // 99 values × 10 tiers

validateTierCapacity(count: number): void
  // throws TierCapacityError if count > 990

computeTierAllocations(gameCount: number, weights?: typeof TIER_WEIGHTS): number[]
  // returns 10-element array; sum === gameCount; all non-negative integers
  // uses largest-remainder method

assignRatings(orderedGameIds: string[], allocations: number[]): Record<string, number>
  // returns integer ratings in [100, 1000]
  // tier 10 first (highest ratings), tier 1 last
  // equal spacing within each tier
  // tier 1 lower bound clamped to 100 (D-11)

applyUpset(winnerId: string, loserId: string, ratings: Record<string, number>): Record<string, number>
  // winner was ranked lower than loser; winner takes loser's rating position
  // games between winner and loser positions shift down one step
  // returns unchanged copy if winner already ranked >= loser

redistribute(ratings: Record<string, number>, weights?: typeof TIER_WEIGHTS): Record<string, number>
  // preserves relative order; recomputes equal spacing for current collection size

initializeRankings(gameIds: string[], weights?: typeof TIER_WEIGHTS): Record<string, number>
  // shuffles gameIds randomly; calls validateTierCapacity; allocates; assigns ratings
```
