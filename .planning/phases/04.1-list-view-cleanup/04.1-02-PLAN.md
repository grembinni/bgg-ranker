    - normalizeUrl helper (module-level function): (url: string) => url.startsWith('//') ? `https:${url}` : url

    RankedGridView main component:
    - Selectors: ratings (Record<string,number>), games (Record<string,Game>), backToComparison, reorderRankedList via useStore
    - State: const [localOrder, setLocalOrder] = useState<string[] | null>(null)
    - State: const [pageOffset, setPageOffset] = useState(0)
    - Ref: const preloadedUrls = useRef<Set<string>>(new Set())
    - sortedFromStore: Object.entries(ratings).sort((a,b) => b[1]-a[1]).map(([id])=>id)
    - orderedIds: localOrder ?? sortedFromStore
    - safeOffset: Math.min(pageOffset, Math.max(0, orderedIds.length - 1)) — clamped per Pitfall 4; use safeOffset instead of pageOffset for all slice operations
    - pageSlice: orderedIds.slice(safeOffset, safeOffset + 100)

    handleDragEnd(event: DragEndEvent):
    - const { active, over } = event
    - if (!over || active.id === over.id) return
    - oldIndex = pageSlice.indexOf(active.id as string)
    - newIndex = pageSlice.indexOf(over.id as string)
    - newPage = arrayMove(pageSlice, oldIndex, newIndex)
    - newFullOrder = [...orderedIds.slice(0, safeOffset), ...newPage, ...orderedIds.slice(safeOffset + pageSlice.length)]
    - setLocalOrder(newFullOrder)
    - reorderRankedList(newFullOrder)

    preloadImages helper (inside component or as module function):
    - Accepts urls: string[]
    - For each url: skip if falsy or already in preloadedUrls.current; add to set; construct new Image(), set .src = normalizeUrl(url)
    - Fire-and-forget (D-13) — no callbacks, no loading state

    useEffect for preloading (depends on safeOffset):
    - Preload all 100 visible thumbnails: pageSlice.map(id => games[id]?.thumbnail).filter(Boolean) as string[]
    - Preload incoming 50 on page advance (next page): orderedIds.slice(safeOffset + 50, safeOffset + 150).map(id => games[id]?.thumbnail).filter(Boolean) as string[]
    - Both calls go to preloadImages()
    - Dependency array: [safeOffset] — runs on mount and each page change

    Render structure:
    - Outer: <div className="max-w-6xl mx-auto px-4 py-8">
    - Header: <header className="flex items-center justify-between mb-4"> with <h1>Rankings ({orderedIds.length} games)</h1> left and <button onClick={backToComparison}>← Back</button> right; button same classes as RankedListView's Back button
    - Pagination controls (above grid): <div className="flex items-center justify-between mb-3"> containing:

        right: <button onClick={() => setPageOffset(Math.min(safeOffset + 50, Math.max(0, orderedIds.length - 1)))} disabled={safeOffset + 100 >= orderedIds.length}>Next 50 →</button>
    - DndContext (sensors, collisionDetection=closestCenter, onDragEnd=handleDragEnd):
        SortableContext (items=pageSlice, strategy=rectSortingStrategy):

          </div>
